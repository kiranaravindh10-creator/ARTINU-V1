# ARTINU — https://artinu.in
#
# Public marketing pages are crawlable. Everything behind sign-in is not:
# those pages hold personal data, need a session to render anything, and would
# only ever be indexed as an empty shell.

User-agent: *
Allow: /

# ── Authentication ──────────────────────────────────────────────────────────
Disallow: /signin
Disallow: /forgot-password
Disallow: /reset-password
Disallow: /register
Disallow: /account/

# ── Signed-in areas ─────────────────────────────────────────────────────────
# Trailing slashes matter. `Disallow: /space` (no slash) is a prefix match, so
# it also blocked /spaces — the main landing page for space owners, and one of
# the most commercially important URLs on the site.
Disallow: /space/
Disallow: /studio/
Disallow: /console/

# ── API ─────────────────────────────────────────────────────────────────────
Disallow: /api/

# Deliberately NOT disallowed: /*.xml$ and /*.txt$ used to be listed here,
# which blocked the sitemap declared at the bottom of this very file.

Sitemap: https://artinu.in/sitemap.xml
